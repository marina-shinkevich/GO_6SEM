1. Как работает сервер? Пакет webdav берет всю сложную логику на себя. Объект webdav.Handler 
— это готовый диспетчер. Он сам парсит входящие запросы, понимает WebDAV-методы и выполняет физические 
операции на жестком диске в папке, которую мы указали в webdav.Dir("./data"). Также он управляет
 LockSystem — механизмом блокировок, который защищает файлы от одновременной перезаписи разными 
 клиентами (методы LOCK и UNLOCK).

2. Какие HTTP-методы добавляет WebDAV? Преподаватель обязательно спросит про методы, так как это суть протокола. Вы должны назвать основные:

    MKCOL (Make Collection) — аналог mkdir, создает директорию (папку).
    PROPFIND — запрос свойств файла или папки (возвращает дату создания, размер и т.д. в формате XML).
    PROPPATCH — изменение этих свойств.
    COPY и MOVE — копирование и перемещение ресурсов прямо на стороне сервера (не скачивая их клиенту).
     Куда именно перемещать, указывается в специальном HTTP-заголовке Destination.
    Классические GET, PUT и DELETE используются для скачивания, загрузки и удаления файлов соответственно.

3. Как Go-клиент отправляет нестандартные методы? Стандартные функции вроде http.Get()
 или http.Post() здесь не подойдут. Для отправки WebDAV-команд мы используем функцию 
 http.NewRequest(method, url, body). Она позволяет передать методу абсолютно любую строку 
 (например, "MKCOL" или "PROPFIND"). Затем мы отправляем этот сформированный запрос с помощью 
 клиента http.DefaultClient.Do(req).

WebDAV (Web Distributed Authoring and Versioning) — это сетевой протокол и расширение над стандартным 
протоколом HTTP, которое позволяет пользователям совместно редактировать, перемещать, удалять и загружать 
файлы на удаленном веб-сервере.

Если говорить максимально просто: WebDAV превращает удаленный веб-сервер в обычную локальную папку 
или сетевой диск на вашем компьютере.


 ------------SERVER
 package main

import (
	"log"      // Пакет для вывода логов и сообщений в консоль сервера
	"net/http" // Стандартный HTTP-пакет Go для запуска веб-сервера

	"golang.org/x/net/webdav" // Официальный расширенный пакет Go для поддержки протокола WebDAV
)

func main() {
	// 1. Инициализируем WebDAV-обработчик (Handler)
	// Обработчик перехватывает входящие HTTP-методы протокола WebDAV (PROPFIND, MKCOL, MOVE и др.)
	// и превращает их в реальные файловые операции на жестком диске.
	handler := &webdav.Handler{
		// Prefix определяет базовый URL-путь для WebDAV. 
		// Значение "/" означает, что корень диска будет доступен прямо по адресу http://localhost:3000/
		Prefix: "/",

		// FileSystem связывает виртуальные папки WebDAV с реальной директорией на сервере.
		// webdav.Dir("./data") указывает, что все файлы будут физически лежать в папке "data" 
		// в той же директории, откуда запускается программа.
		FileSystem: webdav.Dir("./data"),

		// Поле LockSystem намеренно оставлено пустым (nil).
		// Это означает, что встроенный механизм блокировок файлов отключен.
		// Клиенты могут одновременно читать и перезаписывать одни и те же файлы.
	}

	log.Println("WebDAV server is running on http://localhost:3000")

	// 2. Запускаем стандартный HTTP-сервер на порту 3000.
	// Передаем созданный handler, чтобы он обрабатывал все входящие запросы.
	err := http.ListenAndServe(":3000", handler)
	if err != nil {
		// Если порт занят или произошла другая критическая ошибка, логируем её и завершаем программу
		log.Fatal(err) 
	}
}

package main

import (
	"bytes"     // Пакет для работы со срезами байт (нужен для отправки текстовых/XML данных в теле запроса)
	"fmt"       // Пакет для форматированного вывода результатов в консоль
	"io"        // Интерфейсы для ввода-вывода (чтение потоков данных)
	"log"       // Пакет для логирования ошибок
	"net/http"  // Стандартный HTTP-клиент Go для отправки сетевых запросов
)

// Константы для настройки адресов и имен папок
const (
	baseURL    = "http://localhost:3000/" // Корневой адрес WebDAV-сервера (источник)
	destURL    = "http://localhost:3000/" // Адрес назначения (нужен для заголовка Destination)
	baseFolder = "A"                      // Имя первой (базовой) папки
	workFolder = "B"                      // Имя второй (рабочей) папки
)

// request — универсальная обертка для выполнения любых HTTP и WebDAV запросов.
// Принимает: HTTP-метод, целевой URL, тело запроса (body) и словарь с HTTP-заголовками (h).
func request(method, url string, body io.Reader, h map[string]string) {
	// Инициализируем объект HTTP-запроса. 
	// Стандартная библиотека Go позволяет передавать абсолютно любую строку в качестве метода, 
	// что идеально подходит для нестандартных методов WebDAV (MKCOL, PROPFIND и т.д.).
	req, err := http.NewRequest(method, url, body)
	if err != nil {
		log.Println(method, "error during request creation:", err)
		return
	}

	// Итерируемся по переданному словарю заголовков и добавляем их в запрос.
	// Заголовки критически важны для WebDAV (например, Depth или Destination).
	for k, v := range h {
		req.Header.Set(k, v)
	}

	// Выполняем HTTP-запрос с помощью стандартного клиента Go
	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		log.Println(method, "execution error:", err)
		return
	}
	// Обязательно закрываем тело ответа после завершения работы функции, чтобы избежать утечки ресурсов (памяти/дескрипторов)
	defer resp.Body.Close()

	// Вычитываем все байты из потока ответа сервера
	b, _ := io.ReadAll(resp.Body)
	
	// Красиво выводим лог операции в консоль для анализа результатов
	fmt.Println("===", method, url, "===")
	fmt.Println("status:", resp.Status) // Выведет числовой код и текстовое описание (например, "201 Created")
	fmt.Println(string(b))             // Выведет тело ответа (текст, XML-структуру или пустоту)
	fmt.Println()
}

func main() {
	// 1. MKCOL (Make Collection) — Создание папки "A" на сервере.
	// Метод эквивалентен команде `mkdir` в консоли. 
    Они означают, что для создания папки серверу не нужно передавать тело запроса (данные) и не нужно 
    устанавливать дополнительные HTTP-заголовки.
	request("MKCOL", baseURL+baseFolder+"/", nil, nil)

	// 2. PUT — Загрузка файла "hello.txt" внутрь только что созданной папки "A".
	//bytes.NewBuffer преобразует обычную строку в поток io.Reader, пригодный для отправки по сети.
	request("PUT", baseURL+baseFolder+"/hello.txt",
		bytes.NewBuffer([]byte("Hello WebDAV")), // Содержимое (текст) будущего файла
		map[string]string{"Content-Type": "text/plain"},
	)

	// 3. GET — Стандартное скачивание файла по HTTP для проверки, что он успешно записался.
	request("GET", baseURL+baseFolder+"/hello.txt", nil, nil)

	// 4. PROPFIND (Find Properties) — Запрос метаданных (свойств) папки "A".
	// В теле запроса передается XML-инструкция <allprop/>, требующая вернуть ВСЕ свойства объектов.
	// Заголовок "Depth: 1" указывает серверу вернуть свойства самой папки И всех файлов, находящихся строго внутри неё.
	request("PROPFIND", baseURL+baseFolder+"/",
		bytes.NewBuffer([]byte(`<?xml version="1.0"?><propfind xmlns="DAV:"><allprop/></propfind>`)),
		map[string]string{"Depth": "1", "Content-Type": "application/xml"},
	)

	// 5. PROPPATCH (Property Patch) — Изменение или добавление кастомных свойств (метаданных) файла.
	// Мы передаем XML, который устанавливает (<set>) для файла новое свойство <description> со значением "test file".
	request("PROPPATCH", baseURL+baseFolder+"/hello.txt",
		bytes.NewBuffer([]byte(`<?xml version="1.0"?><propertyupdate xmlns="DAV:"><set><prop><description>test file</description></prop></set></propertyupdate>`)),
		map[string]string{"Content-Type": "application/xml"},
	)

	// 6. MKCOL — Создаем вторую, пустую папку "B" для демонстрации операций копирования и перемещения.
	request("MKCOL", baseURL+workFolder+"/", nil, nil)

	// 7. COPY — Копирование файла из папки "A" в папку "B".
	// URL-источник передается как основной адрес запроса.
	// URL-назначения (куда копировать и как назвать файл) обязательно передается в специальном заголовке "Destination".
	request("COPY", baseURL+baseFolder+"/hello.txt",
		nil,
		map[string]string{"Destination": destURL + workFolder + "/hello_copy.txt"},
	)

	// 8. MOVE — Перемещение или переименование файла.
	// Физически берет файл "hello_copy.txt" из папки "B", переносит его в папку "A" под новым именем "hello_moved.txt".
	// Исходный файл в папке "B" после этой операции удаляется сервером автоматически.
	request("MOVE", baseURL+workFolder+"/hello_copy.txt",
		nil,
		map[string]string{"Destination": destURL + baseFolder + "/hello_moved.txt"},
	)

	// 9. DELETE — Очистка сервера. 
	// Каскадно удаляет папки "A" и "B" вместе со всем их содержимым (всеми созданными и перемещенными файлами).
	request("DELETE", baseURL+baseFolder+"/", nil, nil)
	request("DELETE", baseURL+workFolder+"/", nil, nil)
}
