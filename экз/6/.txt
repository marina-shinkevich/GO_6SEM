1. Что такое JSON-RPC и чем он отличается от REST?
 В REST мы обращаемся к разным URL для разных действий (например, GET /users, POST /users). 
 В JSON-RPC (Remote Procedure Call) мы всегда шлем POST-запрос на один и тот же адрес (в нашем случае /rpc). 
 Чтобы сервер понял, что мы хотим сделать, мы передаем в JSON-теле запроса имя функции (поле "method": "sum")
  и её аргументы (поле "params").

2. Зачем нужен тип данных json.RawMessage? Это очень частый вопрос.
 В структуре запроса RPCRequest параметры объявлены как json.RawMessage. 
 Это нужно потому, что заранее неизвестно, какие параметры пришлет клиент — это может быть объект
  {"X":5, "Y":3}, а может быть массив  (для другого метода). json.RawMessage говорит парсеру Go: 
  "Пока не разбирай эту часть JSON, просто сохрани её как сырой текст/байты". Разбор происходит позже,
   уже внутри конкретного метода sum, когда мы точно знаем структуру параметров SumArg.

3. Что такое Notification (уведомление)? В протоколе JSON-RPC, если клиент присылает запрос, 
в котором отсутствует поле "id" (или оно равно nil), это означает, что клиент просто уведомляет 
сервер о событии и не ждет никакого ответа. В обработчике rpcHandler это учитывается: if rq.ID ==
 nil { callMethod(rq) } — мы вызываем функцию, но результат клиенту не отправляем.

4. Зачем в коде проверка raw == '['? Это отличительная особенность стандарта JSON-RPC 2.0 — поддержка 
Batch-запросов (пакетных запросов). Клиент может за один HTTP-запрос прислать сразу массив команд 
(например, сложить, умножить и разделить разные числа). Если сырой JSON начинается с квадратной скобки
 [, сервер понимает, что прилетел массив запросов, и переходит в режим цикла, обрабатывая каждую команду
  по очереди, а в ответ также возвращает массив ответов.

5. Как сервер определяет, какой код выполнять? Мы используем паттерн "Словарь функций" (Dispatch table).
 У нас есть глобальная переменная methods (массив типа map), где ключом выступает строка (имя метода "sum"), а значением — сама анонимная функция func(RPCRequest) RPCResponse. Метод register добавляет функции в этот словарь, а диспетчер callMethod ищет нужную функцию по имени, присланному клиентом, и если не находит — возвращает стандартную ошибку "method not found"



 package main

import (
	"encoding/json" // Пакет для кодирования и декодирования JSON (запросы/ответы)
	"log"           // Пакет для вывода логов в консоль сервера
	"net/http"      // Пакет для запуска веб-сервера и работы с HTTP

	"github.com/gorilla/mux" // Маршрутизатор для удобной настройки путей (роутов)
)

// === 1. ГЛОБАЛЬНЫЕ ПЕРЕМЕННЫЕ И СТРУКТУРЫ ===

var (
	// Глобальная карта (map) для регистрации RPC-методов.
	// Ключ — имя метода (строка), значение — функция-обработчик, принимающая запрос и возвращающая ответ.
	// МЬЮТЕКС УБРАН: карта инициализируется один раз при старте сервера в функции main.
	methods = map[string]func(RPCRequest) RPCResponse{} 
)

// RPCRequest — структура входящего запроса по спецификации JSON-RPC 2.0
type RPCRequest struct {
	Jsonrpc string `json:"jsonrpc"` // Версия протокола (всегда "2.0")
	Method  string `json:"method"`  // Имя вызываемого удаленного метода (например, "sum")
	
	// json.RawMessage позволяет заморозить (отложить) парсинг параметров.
	// Мы десериализуем их позже, когда поймем по полю Method, какая структура данных там лежит.
	Params json.RawMessage `json:"params"`
	ID     interface{}     `json:"id"` // Уникальный идентификатор запроса (число или строка)
}

// RPCResponse — структура стандартного ответа JSON-RPC 2.0

Запись вида json:"jsonrpc" в Go называется структурным тегом (Struct Tag).

Она нужна для того, чтобы связать (смаппить) имя поля в коде Go с именем ключа в текстовом файле JSON, 
который присылает клиент.
type RPCResponse struct {
	Jsonrpc string      `json:"jsonrpc"`
	Result  interface{} `json:"result,omitempty"` // omitempty: если поле пустое, оно не попадёт в JSON-ответ
	Error   interface{} `json:"error,omitempty"`  // Поле для описания ошибок (заполняется, если Result пуст)
	ID      interface{} `json:"id"`               // Возвращаем тот же ID, что прислал клиент
}

// Структуры для конкретного бизнес-метода "sum"
type SumArg struct{ X, Y float64 }     // Параметры, которые ждем от клиента
type SumResult struct{ Value float64 } // Результат, который вернем клиенту

// === 2. ЯДРО RPC-СЕРВЕРА ===

// register — функция записи (регистрации) нового метода в глобальную карту.
func register(name string, fn func(RPCRequest) RPCResponse) {
	methods[name] = fn
}

// callMethod — диспетчер, который ищет функцию в карте по её текстовому имени.
func callMethod(r RPCRequest) RPCResponse {
	// Ищем функцию в глобальной карте `methods`.
	// Чтение из map параллельными горутинами БЕЗОПАСНО, если в этот момент в карту никто ничего не пишет.
	if method := methods[r.Method]; method != nil {
		return method(r) // Вызываем найденную функцию бизнес-логики
	}
	
	// Если клиент передал имя метода, которого нет в нашей карте
	return RPCResponse{
		Jsonrpc: "2.0",
		Error:   "method not found",
		ID:      r.ID,
	}
}

// rpcHandler — единая точка входа (endpoint) для обработки всех входящих HTTP-запросов
func rpcHandler(w http.ResponseWriter, r *http.Request) {
	var raw json.RawMessage

	// Считываем сырое тело HTTP-запроса
    Эта строчка одновременно читает входящие данные от клиента и проверяет, что они написаны в правильном формате JSON.
	if err := json.NewDecoder(r.Body).Decode(&raw); err == nil {

		// Проверка на пакетный (batch) запрос: JSON-RPC позволяет прислать массив запросов в [ ]
		if len(raw) > 0 && raw[0] == '[' {
			// Логика пакетной обработки (упрощена для лабораторной работы)
		} else { 
			// Одиночный запрос
			var rq RPCRequest
			json.Unmarshal(raw, &rq) // Эта строчка переводит (декодирует) текстовый 
            JSON-запрос в готовую Go-структуру, с которой дальше можно работать в коде.

			if rq.ID == nil { 
				// Если ID отсутствует — это Notification (уведомление). 
				// Мы просто выполняем метод, но ничего не пишем в HTTP-ответ.
				callMethod(rq)
			} else { 
				// Обычный запрос, на который клиент ждет ответ.
				response := callMethod(rq) // Маршрутизируем запрос к нужной функции
				
				w.Header().Set("Content-Type", "application/json") // Указываем формат JSON
				json.NewEncoder(w).Encode(response)                // Отправляем JSON-ответ клиенту
			}
		}
	} else {
		// Если клиент прислал "сломанный" JSON, который в принципе не парсится
		json.NewEncoder(w).Encode(RPCResponse{Jsonrpc: "2.0", Error: "json error"})
	}
}

// === 3. ТОЧКА ВХОДА ===

func main() {
	// ШАГ 1: Регистрация методов (Инициализация).
	// Происходит строго ДО запуска веб-сервера. Мы монопольно пишем в карту `methods` из главного потока.
	register("sum", func(r RPCRequest) RPCResponse {
		log.Println("Вызван метод sum")
		var p SumArg

		// Достаем параметры конкретно для метода sum из отложенного хранилища Params
		json.Unmarshal(r.Params, &p)

		// Возвращаем ответ со сложением чисел
		return RPCResponse{
			Jsonrpc: "2.0",
			Result:  SumResult{Value: p.X + p.Y},
			ID:      r.ID,
		}
	})

	// ШАГ 2: Настройка HTTP-маршрутизации.
	router := mux.NewRouter()

	// По спецификации JSON-RPC все операции идут через один URI (эндпоинт) методом POST
	router.HandleFunc("/rpc", rpcHandler).Methods("POST")

	log.Println("JSON-RPC Server started on port 3000")
	
	// Запуск сервера. На каждый запрос http.ListenAndServe начнет плодить параллельные горутины.
	http.ListenAndServe(":3000", router)
}
