1. Как браузер понимает, что файл нужно скачать, а не показать? По умолчанию, если сервер отправит 
картинку (JPG) или текстовый файл (TXT), браузер просто откроет её в новой вкладке. Чтобы принудительно 
вызвать диалоговое окно скачивания файла, мы устанавливаем специальные HTTP-заголовки:

    w.Header().Set("Content-Disposition", "attachment; filename=...") — слово attachment (вложение) является прямой командой браузеру сохранить файл на диск. Параметр filename указывает, под каким именем по умолчанию сохранится файл у пользователя.
    w.Header().Set("Content-Type", "application/octet-stream") — этот MIME-тип указывает, что мы передаем "сырой" поток байтов.

2. Зачем нужна проверка !info.IsDir()? Функция os.Stat(filePath) проверяет состояние файловой системы.
 Без этой проверки хакер (или просто ошибшийся пользователь) мог бы отправить запрос GET /DFS/static/,
  и в переменную filename попало бы пустое значение. В результате сервер попытался бы "скачать" саму 
  папку static, что вызвало бы внутреннюю ошибку сервера. Метод .IsDir() гарантирует, что мы отдаем клиенту 
  только конкретные файлы.
3. Как работает http.ServeFile? Вы можете отметить преподавателю, что в лекциях не используется ручное
 создание буферов и циклов for для чтения файла, как мы это делали при изучении низкоуровневой работы 
 с файловой системой. Вместо этого вызывается высокоуровневая функция http.ServeFile(w, r, filePath). 
 Она очень "умная":

    Она сама открывает и закрывает файл.
    Она стримит (передает частями) файл клиенту, не загружая его целиком в оперативную память сервера (что критически важно для больших файлов, например видео).
    Она автоматически поддерживает докачку файлов (Range requests), если соединение оборвалось.

P.S. В лекциях также упоминается альтернативный способ раздачи статики целыми папками через http.FileServer
 и http.StripPrefix, но именно для контролируемого скачивания конкретного файла (с нашими заголовками) 
 используется подход с handlerDFS.


http://localhost:3000/DFS/static/rf3.jpg



package main

import (
	"log"      // Пакет для вывода логов работы сервера в консоль
	"net/http" // Стандартный сетевой пакет Go для обработки HTTP-запросов
	"os"       // Пакет для работы с операционной системой (чтение файлов, проверка их наличия)
	"path"     // Пакет для манипуляций с путями в формате URL (всегда через косой слеш "/")
	"path/filepath" // Пакет для работы с путями файловой системы конкретной ОС (Windows, Linux)

	// Подключаем продвинутый маршрутизатор Gorilla Mux
	"github.com/gorilla/mux"
)

// handlerDFS (Download File Server) — функция-обработчик для скачивания файла.
// Ожидает запрос вида: GET http://localhost:3000/DFS/static/PNorton.jpg
func handlerDFS(w http.ResponseWriter, r *http.Request) {
	// ШАГ 1: Задаем изолированную папку на сервере, из которой разрешено скачивать файлы.
	// Точка означает "текущая папка, где запущена программа".
	const baseDir = "./static"

	// ШАГ 2: Достаем динамическую часть URL из маршрутизатора Gorilla Mux.
	// mux.Vars(r) превращает URL в карту (map). Если клиент ввел /DFS/static/photo.png,
	// то vars["filename"] вернет строку "photo.png".
	vars := mux.Vars(r)
	filename := vars["filename"]

	// ШАГ 3: Безопасное склеивание путей.
	// Метод filepath.Join объединяет "./static" и, например, "photo.png".
	// Важно: он автоматически определяет ОС. На Linux получится "./static/photo.png", 
	// а на Windows — ".\static\photo.png". Это предотвращает баги кроссплатформенности.
	filePath := filepath.Join(baseDir, filename)

	// ШАГ 4: Спрашиваем у операционной системы статус файла (Метаданные).
	// os.Stat считывает информацию о файле без открытия самого файла.
	// Возвращает объект `info` и ошибку `err` (если файла нет или заблокирован доступ).
	info, err := os.Stat(filePath)

	// Если ошибки нет (err == nil), значит объект по такому пути успешно найден на диске.
	if err == nil {
		// Проверяем: это файл или папка? 
		// Если клиент схитрил и передал имя папки, `info.IsDir()` вернет true.
		if !info.IsDir() {
			// Метод path.Base очищает имя файла от возможных случайных слешей,
			// гарантируя, что мы отправим клиенту только чистое имя, например, "PNorton.jpg".
			downloadName := path.Base(filename)

			// === КЛЮЧЕВОЙ БЛОК ДЛЯ ПРИНУДИТЕЛЬНОГО СКАЧИВАНИЯ ===
			
			// Заголовок Content-Disposition со значением "attachment" — это приказ для браузера.
			// Вместо того чтобы открыть картинку или PDF-файл прямо в текущей вкладке,
			// браузер обязан показать системное окно "Сохранить файл как..." и скачать его.
			w.Header().Set("Content-Disposition", "attachment; filename="+downloadName)

			// Заголовок application/octet-stream указывает универсальный MIME-тип.
			// Он сообщает клиенту: "Внутри ответа летят сырые бинарные байты неизвестного формата".
			// Браузер не будет пытаться запустить этот файл как скрипт, что обеспечивает безопасность.
			w.Header().Set("Content-Type", "application/octet-stream")
			// ===================================================

			// ШАГ 5: Передача данных в сеть.
			// http.ServeFile — это встроенная высокоэффективная функция Go. 
			// Она сама открывает файл на диске, читает его по кусочкам (буферизованно)
			// и отправляет байты в сеть, контролируя, чтобы оперативная память сервера не переполнялась.
			http.ServeFile(w, r, filePath)
		} else {
			// Если по запрошенному пути на сервере лежит папка, возвращаем статус 400 (Плохой запрос)
			http.Error(w, "not a file", http.StatusBadRequest)
		}
	} else if os.IsNotExist(err) {
		// Если функция os.Stat вернула ошибку, проверяем её специальным методом os.IsNotExist.
		// Если файла физически нет в папке static — возвращаем классическую ошибку 404 Not Found.
		http.Error(w, "file not found", http.StatusNotFound)
	} else {
		// Сюда мы попадем, если файл есть, но у процесса Go нет системных прав на его чтение (ошибка 500)
		http.Error(w, "stat error: "+err.Error(), http.StatusInternalServerError)
	}
}

func main() {
	// Инициализируем маршрутизатор Gorilla Mux
	r := mux.NewRouter()

	// Регистрируем маршрут с динамическим шаблоном {filename}.
	// Роутер перехватит любой GET-запрос, начинающийся на /DFS/static/,
	// и всё, что написано дальше, автоматически упакует в переменную "filename".
	r.HandleFunc("/DFS/static/{filename}", handlerDFS).Methods(http.MethodGet)

	log.Println("Server is running on port 3000")
	// Запускаем веб-сервер на порту 3000 и передаем ему наш роутер `r`
	log.Fatal(http.ListenAndServe(":3000", r))
}
