1. Как в одном запросе комбинируются параметры пути (URL) и тело (Body)? Это стандартный архитектурный 
подход при проектировании REST API.

    Через URL (шаблон {id}) мы указываем, над каким конкретно ресурсом мы совершаем действие (например,
     пользователь с ID 123). Для их извлечения применяется mux.Vars(r).
    Через тело запроса (JSON-объект) мы передаем объемные или структурированные данные, которые нужно 
    применить к этому ресурсу (например, новое имя и возраст). Для их извлечения применяется 
    json.NewDecoder(r.Body).Decode(&in).

2. Зачем здесь используется gorilla/mux, а не стандартный мультиплексор? Стандартный роутер net/http в 
базовой реализации Go не умеет извлекать динамические переменные вроде {id} прямо из середины пути. Вам 
пришлось бы регистрировать префикс /person/, а затем вручную отрезать его от r.URL.Path с помощью функций 
пакета strings. Пакет gorilla/mux делает это за нас элегантно и автоматически, а также сразу ограничивает 
доступ только для POST-запросов (.Methods(http.MethodPost)).
3. Почему потоковое декодирование JSON лучше, чем io.ReadAll? Как и в предыдущих билетах, важно подчеркнуть:
 json.NewDecoder(r.Body).Decode(&in) читает сетевой поток небольшими частями напрямую в структуру. В лекциях 
 показан и другой вариант (с использованием io.ReadAll, чтением всего тела в память и последующим 
 json.Unmarshal), но потоковый вариант безопаснее для сервера, так как защищает память от переполнения 
 при получении огромных JSON-файлов.
4. Зачем в структуру ответа добавлен тег json:"id"? Структура PersonOutput агрегирует данные. Мы объединяем 
строковый id, который пришел к нам из URL от роутера mux, и поля name и age, которые мы вытащили из тела 
запроса. Пакет encoding/json использует эти теги, чтобы правильно сформировать итоговую строку ответа,
 которую увидит клиент.



 package main

import (
	"encoding/json" // Пакет-переводчик. Умеет превращать текст JSON в структуры Go и обратно.
	"log"           // Пакет для вывода системных сообщений в консоль (черное окошко) сервера.
	"net/http"      // Базовый пакет Go для работы с сетью и обработки веб-запросов.

	// Подключаем внешний удобный маршрутизатор (роутер) Gorilla Mux.
	"github.com/gorilla/mux" 
)

// PersonInput — это пустая формочка (шаблон), в которую мы сложим данные, 
// присланные клиентом в теле запроса.
// Теги `json:"name"` связывают наши переменные в Go с ключами в JSON-документе из Postman.
type PersonInput struct {
	Name string `json:"name"` // Сюда запишется текст из ключа "name"
	Age  int    `json:"age"`  // Сюда запишется число из ключа "age"
}

// PersonOutput — это формочка для готового ответа. 
// В неё мы объединим (склеим) ID из ссылки и данные, которые вытащили из JSON.
type PersonOutput struct {
	ID   string `json:"id"`   // Идентификатор, который мы забрали прямо из URL-адреса
	Name string `json:"name"` // Имя, взятое из входящего JSON
	Age  int    `json:"age"`  // Возраст, взятый из входящего JSON
}

// handlerPostPerson — функция-обработчик. Она запускается роутером, 
// когда на сервер приходит нужный POST-запрос.
func handlerPostPerson(w http.ResponseWriter, r *http.Request) {
	
	// ШАГ 1: Забираем изменчивую переменную из самой ссылки (URL).
	// Функция `mux.Vars(r)` потрошит адресную строку и собирает все динамические части в список.
	vars := mux.Vars(r)
	
	// Из созданного списка достаем кусок, который лежал на месте ячейки "{id}".
	// Если клиент вызвал /person/777, то в переменную `id` запишется обычный текст "777".
	id := vars["id"]

	// Создаем пустую переменную на основе нашей структуры-заготовки.
	// Сейчас внутри `in` пустота (Name = "", Age = 0).
	var in PersonInput

	// ШАГ 2: Читаем и расшифровываем JSON, присланный в теле запроса (Body).
	// `json.NewDecoder(r.Body)` — создает специального «робота-читателя», который подключается 
	// напрямую к сетевому потоку и ждет байты от клиента.
	// `.Decode(&in)` — заставляет этого робота прочитать текст, перевести его на язык Go 
	// и записать результат прямо по адресу памяти нашей переменной `in` (знак `&` — это адрес).
	if err := json.NewDecoder(r.Body).Decode(&in); err != nil {
		// Если клиент прислал «сломанный» или кривой JSON (например, забыл кавычку или запятую),
		// этот `if` сработает, и мы вернем ошибку 400 Bad Request (Плохой запрос), остановив функцию.
		http.Error(w, "invalid JSON: "+err.Error(), http.StatusBadRequest)
		return
	}

	// ШАГ 3: Склеиваем данные для финального ответа.
	// Создаем объект ответа `out`. ID мы берем из адресной строки (Шаг 1), 
	// а Name и Age вытаскиваем из уже раскодированной структуры `in` (Шаг 2).
	out := PersonOutput{
		ID:   id,
		Name: in.Name,
		Age:  in.Age,
	}

	// ШАГ 4: Выставляем правильный HTTP-заголовок (паспорт ответа).
	// Мы официально сообщаем браузеру или программе Postman: 
	// «Внимание! Сейчас по сети прилетят не просто буквы, а структурированный JSON-документ в кодировке UTF-8».
	// Благодаря этому Postman красиво раскрасит и отформатирует результат.
	w.Header().Set("Content-Type", "application/json; charset=utf-8")

	// ШАГ 5: Превращаем данные Go обратно в текст JSON и отправляем клиенту.
	// `json.NewEncoder(w)` — создает «робота-писателя», привязанного к сетевому ответу `w`.
	// `.Encode(out)` — берет нашу готовую структуру `out`, кодирует её в текстовую строку JSON 
	// и мгновенно выталкивает обратно в сеть пользователю.
	if err := json.NewEncoder(w).Encode(out); err != nil {
		// Если на этапе кодирования что-то сломалось внутри сервера, отдаем статус 500 (Внутренняя ошибка)
		http.Error(w, err.Error(), http.StatusInternalServerError)
	}
}

func main() {
	// Создаем главный «пульт управления» маршрутами (роутер) из библиотеки Gorilla Mux.
	r := mux.NewRouter()

	// Регистрируем наше правило (маршрут).
	// Строка `/person/{id}` означает: если запрос начинается на /person/, а дальше идет 
	// любой текст или цифры, роутер запомнит этот хвост под именем "id".
	// Метод `.Methods(http.MethodPost)` — это жесткий фильтр. Он говорит, что маршрут 
	// сработает ТОЛЬКО на POST-запросы (когда клиент отправляет нам данные).
	r.HandleFunc("/person/{id}", handlerPostPerson).Methods(http.MethodPost)

	// Печатаем в консоль подсказки для разработчика, чтобы знать, что сервер готов к работе.
	log.Println("Server is running on port 3000")
	log.Println("Test with: POST http://localhost:3000/person/123")

	// Включаем наш сервер на порту 3000 и передаем ему настроенный роутер `r`.
	// Если порт 3000 уже занят другой программой, функция `ListenAndServe` вернет ошибку, 
	// а `log.Fatal` красиво завершит работу приложения и запишет это в логи.
	if err := http.ListenAndServe(":3000", r); err != nil {
		log.Fatal(err)
	}
}
