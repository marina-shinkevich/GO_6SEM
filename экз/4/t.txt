package main

import (
	"fmt"      // Пакет для форматированного вывода текста в консоль
	"log"      // Пакет для логирования ошибок
	"net/http" // Стандартный HTTP-пакет Go для создания базового веб-сервера

	// Подключаем популярную и надежную библиотеку Gorilla WebSocket
	"github.com/gorilla/websocket"
)

// upgrader — это конфигурационная структура. Она отвечает за процесс "Handshake" (рукопожатия)
// и превращает обычное HTTP-соединение в постоянное полнодуплексное WebSocket-соединение.
var upgrader = websocket.Upgrader{
	// CheckOrigin — функция для проверки источника запроса (CORS).
	// Возвращая true, мы разрешаем подключаться к нашему серверу абсолютно с любых сайтов/доменов.
	// Для учебного примера это отлично, но в реальном продакшене здесь должна быть проверка домена.
	CheckOrigin: func(r *http.Request) bool {
		return true
	},
}

// handleConnections — функция-обработчик входящих HTTP-запросов на эндпоинт "/ws".
func handleConnections(w http.ResponseWriter, r *http.Request) {
	// 1. АПГРЕЙД (Смена протокола).
	// Метод Upgrade проверяет HTTP-заголовки клиента и переключает сетевой протокол с HTTP на WebSocket.
	// Переменная `ws` становится объектом сетевого соединения, через которое мы будем общаться с клиентом.
	ws, err := upgrader.Upgrade(w, r, nil)
	if err != nil {
		log.Println("Ошибка при апгрейде соединения:", err)
		return // Если апгрейд не удался, закрываем функцию. Клиент получит ошибку.
	}
	// defer гарантирует, что сетевое соединение закроется в любом случае, 
	// когда выполнение функции handleConnections завершится (например, при выходе из цикла)
	defer ws.Close()
	
	fmt.Println("Клиент успешно подключился по WebSocket!")

	// 2. БЕСКОНЕЧНЫЙ ЦИКЛ ОБРАБОТКИ СООБЩЕНИЙ.
	// Так как WebSocket — это постоянное соединение, мы запускаем цикл, который работает 
	// до тех пор, пока клиент сам не закроет вкладку или не произойдет сбой сети.
	for {
		// Чтение сообщения от клиента.
		// Функция ReadMessage блокирует (останавливает) выполнение цикла и "ждет", пока от клиента придут байты.
		// Возвращает: 
		// - messageType: тип сообщения (обычно 1 — текстовый формат, или 2 — бинарный файл)
		// - p: срез байт ([]byte) — само текстовое наполнение сообщения
		// - err: ошибку, если клиент отключился
		messageType, p, err := ws.ReadMessage()
		if err != nil {
			// Сюда программа зайдет, если пользователь закрыл браузер или нажал "Disconnect"
			log.Println("Клиент отключился или произошла ошибка:", err)
			break // Выходим из бесконечного цикла, после чего defer закроет соединение окончательно
		}

		// Вывод полученного сообщения в консоль сервера.
		// Переводим массив байт `p` в обычную читаемую Go-строку с помощью `string(p)`.
		fmt.Printf("Получено от клиента: %s\n", string(p))

		// ОТПРАВКА СООБЩЕНИЯ ОБРАТНО (Эхо-эффект).
		// Берем тот же тип сообщения и те же байты `p` и отправляем их обратно клиенту.
		err = ws.WriteMessage(messageType, p)
		if err != nil {
			log.Println("Ошибка при отправке сообщения клиенту:", err)
			break // Если отправить не удалось (сеть оборвалась), прерываем цикл
		}
	}
}

func main() {
	// Регистрируем наш обработчик на URL-путь "/ws"
	http.HandleFunc("/ws", handleConnections)

	fmt.Println("WebSocket эхо-сервер запущен на http://localhost:3001/ws")
	
	// Запускаем веб-сервер на порту 3001
	err := http.ListenAndServe(":3001", nil)
	if err != nil {
		// Если порт уже занят другой программой, выводим фатальную ошибку и завершаем приложение
		log.Fatal("Ошибка запуска сервера: ", err)
	}
}

<body>

    <h2>Тестирование WebSocket-сервера</h2>
    <div id="messages"></div>
    
    <input type="text" id="messageInput" placeholder="Введите сообщение...">
    <button onclick="sendMessage()">Отправить</button>

    <script>
        // Инициализируем объект встроенного в браузер Web API класса WebSocket.
        // Передаем URL-адрес сервера. Обратите внимание на схему протокола "ws://" вместо "http://".
        // Браузер немедленно отправляет HTTP-запрос для выполнения Handshake (рукопожатия).
        const socket = new WebSocket("ws://localhost:3001/ws");
        
        // Кэшируем DOM-элемент окна чата для последующего добавления в него текста
        const messagesDiv = document.getElementById("messages");

        // === РЕГИСТРАЦИЯ ОБРАБОТЧИКОВ СОБЫТИЙ (Event Listeners) ===

        // Событие onopen: срабатывает, когда сервер ответил статусом 101 и TCP-соединение 
        // успешно переключилось на полнодуплексный протокол WebSocket.
        socket.onopen = function(e) {
            logMessage("[Система] Соединение установлено");
        };

        // Событие onmessage: срабатывает асинхронно каждый раз, когда от сервера прилетает новый фрейм данных.
        // Объект `event` содержит метаданные, а свойство `event.data` — это строка или бинарный объект, присланный сервером.
        socket.onmessage = function(event) {
            logMessage(`[Сервер]: ${event.data}`);
        };

        // Событие onclose: вызывается, если соединение было разорвано или штатно закрыто.
        // Параметр event.wasClean (boolean) указывает, произошло ли закрытие по обоюдному согласию (true)
        // или из-за внезапного сбоя/падения процесса сервера (false).
        socket.onclose = function(event) {
            if (event.wasClean) {
                // event.code содержит числовой статус закрытия (например, 1000 — нормальное закрытие)
                logMessage(`[Система] Соединение закрыто чисто, код=${event.code}`);
            } else {
                logMessage('[Система] Соединение прервано');
            }
        };

        // Событие onerror: срабатывает при возникновении сетевых ошибок (например, если указан неверный порт)
        socket.onerror = function(error) {
            logMessage(`[Ошибка]: ${error.message}`);
        };

        // === БИЗНЕС-ЛОГИКА ИНТЕРФЕЙСА ===

        // Функция отправки сообщения, привязанная к кнопке.
        function sendMessage() {
            const input = document.getElementById("messageInput");
            
            // Валидация: отправляем данные только если текстовое поле не пустое
            if (input.value) {
                // Метод socket.send() инкапсулирует строку в текстовый фрейм WebSocket 
                // и мгновенно отправляет её в открытый TCP-канал связи с сервером.
                socket.send(input.value);
                
                // Отображаем отправленный текст в нашем окне логов локально
                logMessage(`[Вы]: ${input.value}`);
                
                // Очищаем поле ввода для удобства написания следующего сообщения
                input.value = "";
            }
        }

        // Вспомогательная функция для рендеринга логов внутри интерфейса (DOM-манипуляции)
        function logMessage(message) {
            // Создаем новый HTML-тег <div> динамически в памяти браузера
            const messageElement = document.createElement("div");
            
            // Записываем текст сообщения. Используем textContent для безопасности, 
            // чтобы избежать XSS-атак (злоумышленник не сможет внедрить тег <script> в чат)
            messageElement.textContent = message;
            
            // Вставляем созданный div в конец общего блока сообщений
            messagesDiv.appendChild(messageElement);
            
            // Автоматическая прокрутка скроллбара вниз. 
            // Приравниваем позицию скролла (scrollTop) к максимальной высоте содержимого (scrollHeight).
            messagesDiv.scrollTop = messagesDiv.scrollHeight;
        }
    </script>
</body>
