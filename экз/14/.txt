1. Зачем нужен пакет gorilla/mux? Чем он лучше стандартного net/http? Стандартный роутер (мультиплексор)
 Go умеет направлять запросы только по путям (URL), например /D. Из-за этого разработчику приходится вручную
  писать конструкции switch r.Method или if r.Method != "POST" прямо внутри функции-обработчика (как в задании
   №24). Пакет gorilla/mux позволяет делать декларативную маршрутизацию. Цепочка вызовов r.HandleFunc("/D",
    handlerD).Methods(http.MethodPost) автоматически отсечет все GET, PUT или DELETE запросы еще на подступах
     к функции. Если клиент отправит GET-запрос на этот URL, mux сам ответит ошибкой 405 Method Not Allowed, 
     и функция handlerD даже не запустится.


2. Почему в http.ListenAndServe передается r вместо nil? В предыдущих задачах мы писали 
http.ListenAndServe(":3000", nil). Значение nil говорило серверу использовать встроенный глобальный роутер 
(DefaultServeMux). Создав r := mux.NewRouter(), мы сделали свой собственный, независимый роутер. Чтобы сервер 
начал направлять запросы именно через него, мы обязаны явно передать объект r вторым параметром в функцию 
запуска сервера.

3. Использование map[string]interface{} для формирования JSON В предыдущей задаче мы жестко задавали 
структуру Output для отправки ответа. Однако на лекциях показан и другой подход: использование словаря 
map[string]interface{}. Тип interface{} в Go означает "абсолютно любой тип данных". Это позволяет "на лету"
 собирать JSON-ответы сложной вложенности, не создавая под них отдельные структуры. Как видно в коде, мы 
 положили в этот словарь и простые строки (r.Header.Get), и целые числа (data.X), и даже вложили целиком 
 другую структуру (data) — пакет encoding/json прекрасно справится с сериализацией такой солянки.


 package main

import (
	"encoding/json" // Пакет-переводчик. Нужен для конвертации текста JSON в объекты Go и обратно.
	"fmt"           // Пакет для текстового форматирования и склеивания строк.
	"log"           // Пакет для логирования и вывода сообщений в консоль (черное окошко) сервера.
	"net/http"      // Главный сетевой пакет Go для работы с протоколом HTTP.

	// Подключаем внешний продвинутый роутер (маршрутизатор) Gorilla Mux
	"github.com/gorilla/mux" 
)

// RequestD — это пустая формочка-шаблон, в которую сервер аккуратно разложит 
// текстовые данные JSON, пришедшие от клиента в теле POST-запроса.
// Теги `json:"..."` связывают наши переменные в Go (с большой буквы) с ключами из Postman (с маленькой).
type RequestD struct {
	Op string `json:"op"` // Сюда запишется строка с операцией (например, "add")
	X  int    `json:"x"`  // Сюда запишется первое число
	Y  int    `json:"y"`  // Сюда запишется второе число
}

// handlerD — функция-обработчик. Она автоматически включается роутером, 
// когда на адрес "/D" прилетает правильный POST-запрос.
func handlerD(w http.ResponseWriter, r *http.Request) {
	// Создаем пустую переменную на основе нашей структуры.
	// Сейчас внутри data пустота: Op = "", X = 0, Y = 0.
	var data RequestD

	// ШАГ 1: Читаем и расшифровываем JSON из сетевого кабеля (тела запроса r.Body).
	В этой строчке кода вы декодируете текст JSON именно в объект (а если говорить точными 
    терминами языка Go — в экземпляр структуры RequestD).(заполнение объекта data данными)
	if err := json.NewDecoder(r.Body).Decode(&data); err != nil {
		// Если клиент прислал сломанный JSON (например, забыл запятую), этот if сработает.
		// Сервер вернет ошибку 400 Bad Request (Плохой запрос) и сразу прервет выполнение функции.
		http.Error(w, "invalid JSON", http.StatusBadRequest)
		return
	}

	// ШАГ 2: Бизнес-логика. Собираем данные для обратного ответа.
	// Мы используем универсальный тип `map[string]interface{}`. 
	// Это гибкий словарь, где ключи — всегда строки (`string`), 
	// а значения могут быть вообще ЛЮБОГО типа (`interface{}`) — и числами, и текстом, и целыми структурами.
	resp := map[string]interface{}{
		"received":    data, // Закладываем внутрь ответа всю полученную структуру `data` целиком
		"op":          data.Op,
		"x":           data.X,
		"y":           data.Y,
		"contentType": r.Header.Get("Content-Type"), // Запоминаем, в каком формате клиент слал нам данные
		"info":        fmt.Sprintf("%s %s", r.Method, r.RequestURI), // Склеиваем тип метода и адрес запроса
	}

	// ШАГ 3: Заполняем «паспорт» нашего ответа (HTTP-заголовок).
	// Официально сообщаем программе Postman или браузеру:
	// «Внимание! Мы возвращаем структурированный JSON-документ в кодировке UTF-8».
	w.Header().Set("Content-Type", "application/json; charset=utf-8")

	// ШАГ 4: Превращаем наш словарь Go обратно в текст JSON и выталкиваем в сеть клиенту.
	// `json.NewEncoder(w)` — создает «робота-писателя», подключенного к каналу ответа `w`.
	// `.Encode(resp)` — берет карту `resp`, кодирует её в JSON-строку и мгновенно отправляет пользователю.
	if err := json.NewEncoder(w).Encode(resp); err != nil {
		// Если на этапе кодирования внутри сервера произошел сбой, возвращаем статус 500 (Внутренняя ошибка)
		http.Error(w, err.Error(), http.StatusInternalServerError)
	}
}

func main() {
	// Создаем главный «пульт управления» маршрутами (роутер) из библиотеки Gorilla Mux.
	r := mux.NewRouter()

	// Настраиваем правило для роутера:
	// Если кто-то стучится на точный адрес "/D", и делает это строго методом POST, 
	// то роутер должен мгновенно запустить нашу функцию handlerD.
	r.HandleFunc("/D", handlerD).Methods(http.MethodPost)

	log.Println("Server is running on", 3000)

	// Включаем сервер на локальном адресе 127.0.0.1 и порту 3000.
	// Важно: вторым параметром мы отдаем наш настроенный пульт `r`. 
	// Если порт 3000 уже занят другой программой, log.Fatal аварийно выключит сервер и выведет ошибку.
	if err := http.ListenAndServe("127.0.0.1:3000", r); err != nil {
		log.Fatal(err)
	}
}
