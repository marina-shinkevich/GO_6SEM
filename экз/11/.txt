1. Что такое Path-параметры и как их достать? Это переменные, зашитые прямо в путь адреса, например 
http://localhost:3000/P1/555. Для их извлечения мы используем функцию mux.Vars(r). Она анализирует входящий 
запрос и возвращает ассоциативный массив (словарь), где ключами являются названия переменных из наших фигурных
 скобок (например, "id"), а значениями — то, что фактически ввел пользователь.

2. Зачем нужна функция strconv.Atoi()? Преподаватель может спросить: "Почему нельзя сразу использовать
 переменную idStr в расчетах?". В протоколе HTTP все данные передаются текстом. Функция mux.Vars тоже 
 всегда возвращает массив строк. Если нам нужно использовать id для поиска в базе данных (где первичный 
 ключ имеет тип int), мы обязаны сначала сконвертировать строку в число. Для этого применяется функция Atoi
  (ASCII to Integer) из стандартного пакета strconv.

3. Как работают шаблоны с регулярными выражениями {id:+}? Это мощнейший механизм защиты (валидации) 
приложения на уровне роутинга. Если вы зарегистрировали маршрут r.HandleFunc("/G/{id:+}", handlerG), 
а пользователь в браузере введет http://localhost:3000/G/XXX (то есть передаст буквы вместо цифр), то 
роутер gorilla/mux проверит регулярное выражение, поймет, что оно не совпадает, и даже не станет запускать 
ваш обработчик. Клиенту мгновенно и автоматически вернется стандартная ошибка 404 page not found. Э
то избавляет программиста от написания проверок «является ли введенный текст числом» внутри самой функции.


package main

import (
	"fmt"      // Пакет для отправки текстовых ответов обратно клиенту (в Postman/браузер)
	"log"      // Пакет для вывода логов и системных сообщений в консоль сервера
	"net/http" // Стандартный сетевой пакет Go для работы по протоколу HTTP
	"strconv"  // Пакет для конвертации типов (в данном случае: превращаем текст в целое число)

	// Подключаем продвинутый маршрутизатор (роутер) Gorilla Mux
	"github.com/gorilla/mux"
)

// handlerP1 — обрабатывает маршрут, в котором передается один параметр прямо в пути URL.
// Пример успешного запроса из Postman: GET http://localhost:3000/P1/123
func handlerP1(w http.ResponseWriter, r *http.Request) {
	// ШАГ 1: Извлекаем переменные пути.
	// Функция `mux.Vars(r)` сканирует строку адреса, находит там динамические части 
	// и упаковывает их в ассоциативный массив (словарь) формата map[string]string.
	vars := mux.Vars(r)

	// ШАГ 2: Достаем строковое значение параметра по ключу "id"
	// Ключ "id" совпадает с тем именем, которое мы указали в шаблоне в функции main().
	idStr := vars["id"]

	// ШАГ 3: Конвертация типов (Текст -> Число).
	// Все данные из URL-адреса изначально приходят в виде строк (string).
	// Чтобы превратить строку в полноценное целое число, мы используем функцию `strconv.Atoi`.
	// Знак нижнего подчеркивания `_` игнорирует переменную ошибки, так как роутер 
	// на этапе проверки регулярного выражения гарантирует нам, что там будут только цифры.
	id, _ := strconv.Atoi(idStr)

	// ШАГ 4: Формируем и отправляем ответ.
	// Спецификатор `%d` используется для форматированного вывода целых чисел (int).
	fmt.Fprintf(w, "id = %d\n", id)
}

// handlerP2 — обрабатывает сложный маршрут, где в пути передаются сразу два параметра.
// Пример успешного запроса из Postman: GET http://localhost:3000/P2/123/student
func handlerP2(w http.ResponseWriter, r *http.Request) {
	// Точно так же считываем все динамические параметры из входящего HTTP-запроса
	vars := mux.Vars(r)

	// Забираем значения двух параметров по их уникальным именам (ключам)
	idStr := vars["id"] // Сюда попадет числовой сегмент пути
	sStr := vars["s"]   // Сюда попадет текстовый сегмент (например, "student")

	// Переводим строковый ID в системное число типа int
	id, _ := strconv.Atoi(idStr)

	// Отправляем клиенту комбинированный ответ, содержащий и число (%d), и строку (%s)
	fmt.Fprintf(w, "id = %d, s = %s\n", id, sStr)
}

func main() {
	// Создаем новый роутер из библиотеки gorilla/mux
	r := mux.NewRouter()

	// МАРШРУТ №1: Фильтрация с помощью регулярного выражения.
	// Конструкция `{id:[0-9]+}` создает переменную пути "id". 
	// Регулярное выражение `[0-9]+` означает: этот маршрут сработает ТОЛЬКО если 
	// на месте ID будет находиться последовательность из цифр от 0 до 9 (минимум одна цифра).
	// Если передать туда буквы (например, /P1/apple), роутер выдаст 404 Page Not Found.
	r.HandleFunc("/P1/{id:[0-9]+}", handlerP1).Methods(http.MethodGet)

	// МАРШРУТ №2: Комбинированный маршрут с несколькими параметрами.
	// Первый параметр `{id:[0-9]+}` жестко валидируется на цифры.
	// Второй параметр `{s}` принимает абсолютно любые текстовые символы до конца строки URL.
	r.HandleFunc("/P2/{id:[0-9]+}/{s}", handlerP2).Methods(http.MethodGet)

	log.Println("Server is running on port 3000")
	
	// Включаем сервер на порту 3000. В качестве обработчика передаем наш роутер `r`.
	// log.Fatal подстрахует программу: если порт 3000 уже занят, сервер красиво завершит работу с ошибкой.
	log.Fatal(http.ListenAndServe(":3000", r))
}



Сервер понимает, что это id, на основе URL-шаблона, зарегистрированного в маршрутизаторе. В строке маршрута 
мы декларативно используем фигурные скобки {id:[0-9]+}, которые задают имя переменной пути и ограничивают её
регулярным выражением. При поступлении запроса роутер сопоставляет сегменты URL-адреса с этим шаблоном, 
автоматически извлекает нужный кусок текста и сохраняет его в контексте запроса под ключом "id", откуда 
мы и забираем его с помощью mux.Vars(r)».
