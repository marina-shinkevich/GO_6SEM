1. Почему используется multipart/form-data, а не application/x-www-form-urlencoded? Если нужно отправить только короткий текст, используется формат urlencoded (он склеивает данные как x=1&y=2). Но если нужно отправить картинку или PDF-файл, такой метод будет крайне неэффективным. Формат multipart/form-data делит тело запроса на отдельные "части" (блоки), каждая из которых имеет свой собственный заголовок (имя поля, тип контента) и отделена специальным маркером — boundary.
2. Что значит магическая запись 10 << 20 в функции ParseMultipartForm? Это побитовый сдвиг влево. Число 10 в двоичном виде сдвигается на 20 позиций влево, что математически эквивалентно умножению на 220. Так как 1 Мегабайт = 1024×1024 байт (220 байт), то 10×220 — это ровно 10 Мегабайт. Это лимит оперативной памяти, который Go выделит для парсинга этой формы. Всё, что превысит 10 МБ, будет безопасно сохранено во временный файл на диске сервера, чтобы сервер не "упал" от нехватки оперативной памяти при загрузке огромных файлов.
3. Как извлекаются файлы? (r.FormFile) Мы вызываем file, header, err := r.FormFile("s").

    file — это интерфейс multipart.File (поток), из которого мы можем вычитывать байты с помощью метода .Read().
    header — это объект с метаданными. Именно из него сервер узнает, как файл назывался на компьютере пользователя (header.Filename) и сколько байт он весит (header.Size).

4. Зачем в коде присутствует BOM (buf[0:2])? На лекционном слайде показано извлечение BOM (Byte Order Mark). Это специальный невидимый символ в самом начале текстовых файлов (например, FF FE для кодировки UTF-16LE), который подсказывает текстовым редакторам, в какой кодировке нужно читать файл. Код вырезает первые два байта буфера и распечатывает их, чтобы проанализировать, есть ли в переданном файле этот маркер.
5. Как протестировать этот код через консоль? На слайде показано, 
что для отправки multipart-запросов используется команда curl с флагом -F
 (от слова Form). Если вы напишете: 
 curl.exe -X POST http://localhost:3000/MPF -F "x=10" -F "y=20" -F "s=@""D:\Уник\ПИС\Экзамен\gorilla, multipart\myText.txt"""
 , утилита curl автоматически прочитает файл myText.txt с вашего диска, сама сгенерирует сложный multipart/form-data с boundary-разделителями и отправит на сервер.

  package main

import (
	"fmt"      // Пакет для отправки текстовых ответов обратно клиенту/браузеру
	"log"      // Пакет для вывода логов работы сервера в консоль
	"net/http" // Стандартный сетевой пакет Go для работы с HTTP-протоколом

	// Первое требование задания: используем продвинутый роутер Gorilla Mux
	"github.com/gorilla/mux"
)

// handlerMPF — функция, которая берет на себя обработку POST-запросов.
// Она умеет разбирать сложный формат multipart/form-data, где одновременно летят и текст, и файлы.
func handlerMPF(w http.ResponseWriter, r *http.Request) {
	
	// ШАГ 1: Парсинг (разбор) multipart-формы.
	// `ParseMultipartForm` заставляет Go прочитать тело запроса и разделить его на отдельные элементы.
	// Конструкция `10 << 20` — это битовый сдвиг, равный ровно 10 Мегабайтам (10 * 1024 * 1024 байт).
	// Это лимит: файлы до 10 МБ сервер обработает прямо в быстрой оперативной памяти (RAM).
	// Если клиент шлет файл весом 50 МБ, сервер не упадет по нехватке памяти, а автоматически 
	// сохранит "излишки" во временные файлы на жестком диске.
	if err := r.ParseMultipartForm(10 << 20); err != nil {
		// Если клиент прислал битые данные или оборвал загрузку — возвращаем статус 400 Bad Request
		http.Error(w, "cannot parse multipart form: "+err.Error(), http.StatusBadRequest)
		return
	}

	// ШАГ 2: Извлечение обычных текстовых полей формы.
	// r.FormValue автоматически заглядывает внутрь разобранной формы и ищет ключи "x" и "y".
	x := r.FormValue("x")
	y := r.FormValue("y")

	// ШАГ 3: Извлечение загруженного файла.
	// r.FormFile ищет в форме поле с именем "s".
	// Возвращает:
	// - file: поток данных для чтения самого файла (io.Reader)
	// - header: объект с метаданными (имя файла на компьютере клиента, размер в байтах)
	// - err: ошибку, если поля "s" в запросе не оказалось
	file, header, err := r.FormFile("s")
	if err != nil {
		http.Error(w, "cannot get file field 's': "+err.Error(), http.StatusBadRequest)
		return
	}
	// Обязательно закрываем дескриптор файла при выходе из функции.
	// Это освобождает оперативную память и временные ресурсы операционной системы.
	defer file.Close()

	// ШАГ 4: Чтение содержимого файла.
	// Создаем "стакан" (буфер) для данных — срез байт размером ровно 100 ячеек.
	buf := make([]byte, 100) 
	
	// file.Read заполняет наш буфер байтами из файла.
	// Переменная `n` запоминает, сколько именно байт удалось прочитать (например, если файл маленький и весит всего 20 байт, n будет равен 20).
	n, err := file.Read(buf) 

	// ШАГ 5: Формирование ответа клиенту.
	// Спецификатор %s выводит текст, %d — числа.
	// %x выводит байты в шестнадцатеричном формате (Hex) — это часто используют для чтения заголовков (BOM).
	// Мы берем кусочек буфера от 0 до 2 байта под BOM, а остаток (от 2 до n) переводим в текст файла.
	fmt.Fprintf(w, "x = %s, y = %s, FileName = %s, FileSize = %d, BOM = %x, File content = %s\n",
		x, y, header.Filename, header.Size, string(buf[0:2]), string(buf[2:n]))
}

func main() {
	// Инициализируем маршрутизатор из библиотеки gorilla/mux
	r := mux.NewRouter()

	// Регистрируем наш путь "/MPF". 
	// Ограничиваем его методом .Methods(http.MethodPost), так как multipart-данные 
	// по стандарту передаются только внутри тела POST-запросов.
	r.HandleFunc("/MPF", handlerMPF).Methods(http.MethodPost)

	log.Println("Server is running on port 3000")
	
	// Запускаем HTTP-сервер на порту 3000 и передаем ему наш роутер `r`
	log.Fatal(http.ListenAndServe(":3000", r))
}
