1. Что такое теги структур (json:"...")? В Go поля структур принято писать с заглавной буквы (чтобы они
 были экспортируемыми и пакет json мог получить к ним доступ). Однако в самом JSON ключи обычно пишутся
  с маленькой буквы (например, "op", "x", "y"). Теги структур — это метаданные, которые указывают пакету
   encoding/json точное правило маппинга (какое поле структуры какому JSON-ключу соответствует).

2. Как работает json.NewDecoder(r.Body).Decode(&in)? Это потоковое (stream) чтение. Преподаватель может 
спросить: "А почему не использовать json.Unmarshal?" (он тоже есть в лекциях). Вы отвечаете: метод Unmarshal
 требует предварительно вычитать весь файл (тело запроса) в оперативную память с помощью io.ReadAll. 
 А связка NewDecoder().Decode() читает данные напрямую из сетевого потока мелкими кусками и сразу преобразует
  в структуру. Это гораздо эффективнее, экономит память и защищает сервер от перегрузки при больших запросах.
   Обратите внимание на амперсанд &in — мы передаем адрес в памяти, куда нужно сложить результат.

3. Как формируется ответ? Логика та же, но в обратную сторону. Вместо того чтобы сначала формировать
 строку в памяти, а потом её печатать, мы используем потоковый json.NewEncoder(w).Encode(out). Он берет
  нашу структуру out, на лету переводит её в JSON-байты и отправляет прямо в сетевой поток ответа w (в
   браузер или curl-клиент).

4. Зачем нужна строка w.Header().Set("Content-Type", "application/json")? По умолчанию сервер Go 
отправляет ответ как обычный текст (text/plain). Устанавливая этот заголовок до начала кодирования, мы явно 
подсказываем клиенту (браузеру или другому приложению), что сейчас прилетят структурированные JSON-данные,
 чтобы клиент мог их правильно обработать.



 package main

import (
	"encoding/json" // Пакет-конвертер. Умеет превращать текстовый JSON из сети в объекты Go и наоборот.
	"log"           // Пакет для красивого вывода сообщений в консоль (черное окошко) сервера.
	"net/http"      // Главный системный пакет Go для работы с сетью и обработки веб-запросов.
)

// Input — это пустая формочка-шаблон, в которую сервер аккуратно разложит 
// текстовые данные JSON, пришедшие от клиента в теле POST-запроса.
// Теги в обратных кавычках `json:"..."` связывают наши переменные в Go (с большой буквы) с ключами из Postman (с маленькой).
type Input struct {
	Op string `json:"op"` // Сюда запишется знак математической операции (например, "+", "-", "*")
	X  int    `json:"x"`  // Сюда запишется первое целое число
	Y  int    `json:"y"`  // Сюда запишется второе целое число
}

// Output — это формочка для готового ответа. 
// В неё мы упакуем исходные числа, знак операции и итоговый посчитанный результат.
type Output struct {
	Op     string `json:"op"`     // Повторяем знак операции для наглядности ответа
	X      int    `json:"x"`      // Повторяем первое число
	Y      int    `json:"y"`      // Повторяем второе число
	Result int    `json:"result"` // Сюда запишется готовый математический ответ калькулятора
}

// handlerJSON — функция-обработчик. Она автоматически включается, 
// когда кто-то присылает запрос на адрес "/JSON".
func handlerJSON(w http.ResponseWriter, r *http.Request) {
	// ШАГ 1: Проверяем тип запроса.
	// Нам нужен строго метод POST (отправка данных). Если клиент пытается просто открыть страницу 
	// через браузер (это метод GET), этот `if` сработает и прогонит его.
	if r.Method != http.MethodPost {
		// Отдаем клиенту ошибку 405 (Метод не поддерживается) и мгновенно завершаем работу функции (`return`).
		http.Error(w, "Метод не поддерживается", http.StatusMethodNotAllowed)
		return
	}

	// Создаем пустую переменную на основе нашей структуры `Input`.
	// Сейчас внутри `in` абсолютная пустота (Op = "", X = 0, Y = 0).
	var in Input

	// ШАГ 2: Читаем и расшифровываем JSON из сетевого потока (тела запроса r.Body).
	// `json.NewDecoder(r.Body)` — создает виртуального «робота-читателя», подключенного к сети.
	// `.Decode(&in)` — заставляет его прочитать текст JSON из Postman и разложить значения прямо 
	// по адресам памяти нашей переменной `in` (знак `&` передает точный адрес памяти).
	if err := json.NewDecoder(r.Body).Decode(&in); err != nil {
		// Если клиент прислал сломанный текст JSON (забыл кавычку или запятую), этот `if` сработает.
		// Сервер вернет ошибку 400 Bad Request (Плохой запрос) и прервет выполнение функции.
		http.Error(w, "invalid JSON: "+err.Error(), http.StatusBadRequest)
		return
	}

	// Заводим переменные для подсчета результата
	var result int
	var isResult bool = true // Флаг: «Всё хорошо, операция нам знакома»

	// ШАГ 3: Бизнес-логика (наш калькулятор).
	// Оператор `switch` смотрит на знак операции, который прислал клиент в `in.Op`.
	switch in.Op {
	case "+":
		result = in.X + in.Y // Если плюс — складываем
	case "-":
		result = in.X - in.Y // Если минус — вычитаем
	case "*":
		result = in.X * in.Y // Если звездочка — умножаем
	default:
		// Если клиент прислал то, чего мы не знаем (например, деление "/" или текст "hello")
		isResult = false // Опускаем флаг: «Эту операцию мы не поддерживаем»
	}

	// ШАГ 4: Формирование ответа в случае успеха
	if isResult {
		// Если операция прошла успешно, заполняем нашу выходную формочку `out`.
		// Дублируем туда исходные данные и дописываем получившийся `result`.
		out := Output{Op: in.Op, X: in.X, Y: in.Y, Result: result}

		// ШАГ 5: Заполняем «паспорт» нашего ответа (HTTP-заголовок).
		// Официально сообщаем программе Postman или браузеру:
		// «Внимание! Мы возвращаем структурированный JSON-документ в кодировке UTF-8».
		w.Header().Set("Content-Type", "application/json; charset=utf-8")

		// ШАГ 6: Превращаем наш объект Go обратно в текст JSON и выталкиваем в сеть клиенту.
		// `json.NewEncoder(w)` — создает «робота-писателя», подключенного к каналу ответа `w`.
		// `.Encode(out)` — берет объект `out`, кодирует его в JSON-строку и отправляет пользователю.
		if err := json.NewEncoder(w).Encode(out); err != nil {
			// Если внутри самого сервера на этапе кодирования JSON что-то сломалось, отдаем статус 500
			http.Error(w, err.Error(), http.StatusInternalServerError)
		}
	} else {
		// ШАГ 7: Если клиент прислал неизвестный знак (флаг isResult стал false).
		// Возвращаем ошибку 400 Bad Request и пишем причину: «unsupported operation» (операция не поддерживается).
		http.Error(w, "unsupported operation", http.StatusBadRequest)
	}
}

func main() {
	// Регистрируем наш маршрут в стандартном роутере Go.
	// Говорим серверу: «Если кто-то стучится по адресу "/JSON", запуск функцию handlerJSON».
	http.HandleFunc("/JSON", handlerJSON)
	
	// Выводим красивую подсказку в консоль сервера, чтобы разработчик знал, по какой ссылке тестировать.
	log.Println("Сервер запущен: http://localhost:3000/JSON")
	
	// Включаем сервер на порту 3000. 
	// Второй параметр `nil` означает, что мы используем стандартный встроенный роутер Go (DefaultServeMux).
	// `log.Fatal` подстрахует программу: если порт 3000 уже занят, сервер красиво закроется с ошибкой в консоли.
	log.Fatal(http.ListenAndServe(":3000", nil))
}
