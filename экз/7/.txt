1. В чем отличие от стандартного роутера net/http? Если бы мы использовали стандартную библиотеку, 
мы могли бы зарегистрировать только сам путь http.HandleFunc("/Q", handler). Нам пришлось бы внутри
 функции писать if и вручную проверять, пустые ли строки возвращает r.URL.Query().Get("x")
 , чтобы решить, что делать дальше. gorilla/mux позволяет вынести эту бизнес-логику на уровень
  диспетчера (роутера). Это делает код намного чище: каждая функция занимается только своей задачей 
  и гарантированно знает, что нужные параметры уже присутствуют.

2. Как работает синтаксис Queries("x", "{x}")? Метод Queries принимает пары строк (ключ-значение).

    Первый аргумент ("x") указывает, какое имя переменной роутер должен искать после знака вопроса ?.
    Второй аргумент ("{x}") — это шаблон значения. В данном случае он означает: «значение может быть
     абсолютно любым, главное, чтобы сам параметр x присутствовал в запросе».

Дополнительный плюс: можно делать жесткую маршрутизацию даже по конкретным значениям. 
Если написать .Queries("mode", "admin"), то маршрут сработает только в том случае, 
если клиент ввел ?mode=admin.
3. Что будет, если параметры не совпадут? Это главное преимущество такого подхода к маршрутизации.
 Если пользователь введет http://localhost:3000/Q?z=100 (передаст какой-то левый параметр z вместо x), 
 роутер mux увидит, что ни один из фильтров .Queries не совпадает, и мгновенно отобьет запрос со статусом 
 404 Page Not Found. Сервер не потратит ресурсы на запуск функций-обработчиков.
4. Зачем важен порядок регистрации маршрутов? Роутер mux проверяет маршруты сверху вниз (в том порядке, 
в котором они написаны в коде). Если бы мы сначала зарегистрировали маршрут Queries("x", "{x}"), а 
под ним Queries("x", "{x}", "y", "{y}"), то запрос ?x=1&y=2 всегда попадал бы в первый обработчик (ведь 
условие наличия x там выполняется). Поэтому в Go существует негласное правило: сначала описываем самые 
специфичные (строгие) маршруты, а затем более общие.



package main

import (
	"fmt"      // Пакет для отправки текстовых ответов обратно в браузер/клиент
	"log"      // Пакет для красивого вывода системных сообщений в консоль сервера
	"net/http" // Стандартный сетевой пакет Go для работы с HTTP-протоколом

	// Подключаем популярный маршрутизатор (роутер) Gorilla Mux
	"github.com/gorilla/mux"
)

// handlerQ_XY — функция-обработчик. Она сработает ТОЛЬКО тогда, 
// когда в адресной строке будут написаны одновременно и параметр 'x', и параметр 'y'.
func handlerQ_XY(w http.ResponseWriter, r *http.Request) {
	// r.URL.Query() — заставляет Go распарсить (разрезать) хвост ссылки после знака "?"
	// .Get("x") — достает конкретное значение, которое привязано к букве "x"
	x := r.URL.Query().Get("x")
	y := r.URL.Query().Get("y")

	// Fprintf отправляет сформированный текст ответа клиенту (в переменную-ответ w)
	fmt.Fprintf(w, "Сработал маршрут XY!\nПараметры: x = %s, y = %s\n", x, y)
}

// handlerQ_X — функция-обработчик. Она сработает, если клиент передал ТОЛЬКО 'x'.
func handlerQ_X(w http.ResponseWriter, r *http.Request) {
	// Достаем значение параметра 'x' из ссылки
	x := r.URL.Query().Get("x")

	fmt.Fprintf(w, "Сработал маршрут только для X!\nПараметр: x = %s\n(параметр y не передан)\n", x)
}

func main() {
	// Создаем умный маршрутизатор (роутер) из библиотеки gorilla/mux
	r := mux.NewRouter()

	// МАРШРУТ №1: Строгий (Проверяет два параметра)
	// Метод .Queries("x", "{x}", "y", "{y}") — это жесткий фильтр. Он говорит роутеру:
	// "Пропусти запрос дальше, только если в URL есть ключ 'x' с любым значением '{x}' 
	// И ключ 'y' с любым значением '{y}'".
	// ПРАВИЛО: Самые строгие и длинные правила мы обязаны регистрировать в коде ПЕРВЫМИ.
	r.HandleFunc("/Q", handlerQ_XY).
		Methods(http.MethodGet).
		Queries("x", "{x}", "y", "{y}")

	// МАРШРУТ №2: Менее строгий (Проверяет только один параметр)
	// Если клиент отправил ссылку, где есть 'x', но забыл указать 'y', 
	// Маршрут №1 его отбросит. Запрос упадет ниже и совпадет с этим правилом.
	r.HandleFunc("/Q", handlerQ_X).
		Methods(http.MethodGet).
		Queries("x", "{x}")

	// Печатаем в консоль подсказки, как правильно тестировать наш сервер
	log.Println("Server is running on port 3000")
	log.Println("Тест 1 (оба параметра):  GET http://localhost:3000/Q?x=10&y=20")
	log.Println("Тест 2 (один параметр):  GET http://localhost:3000/Q?x=10")
	log.Println("Тест 3 (без параметров): GET http://localhost:3000/Q (выдаст ошибку 404)")

	// Включаем сервер на 3000 порту и отдаем ему наш настроенный роутер `r` вместо стандартного `nil`.
	// Если порт занят, log.Fatal аварийно выключит программу и запишет лог.
	log.Fatal(http.ListenAndServe(":3000", r))
}
