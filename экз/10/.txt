1. Что такое Query-параметры? Это классический способ передачи небольших данных на сервер 
(обычно используется для фильтрации, поиска или пагинации). Они добавляются в конец URL после знака 
вопроса ? в формате ключ=значение, а между собой разделяются амперсандом &. В отличие от Path-параметров
 (которые встроены в саму структуру пути /A/123), Query-параметры опциональны по стандартам HTTP.

2. Как работает r.URL.Query().Get("x")? Под капотом Go сам анализирует сырой URL запроса. 
Функция Query() возвращает тип url.Values — это структура, представляющая собой словарь map[string][]string. 
Почему там массив строк? Потому что стандарт HTTP позволяет передать несколько параметров с одинаковым именем 
(например, ?id=1&id=2). Метод .Get("x") — это удобная обертка, которая извлекает именно первое переданное
 значение, а если параметра вообще нет, она безопасно возвращает пустую строку, не вызывая падения программы
  (panic).


3. В чем главная фишка .Queries(...) в пакете mux? Как видно на лекционных слайдах, 
gorilla/mux умеет осуществлять маршрутизацию не только по URL и HTTP-заголовкам, но и по query-параметрам
. Если вы написали .Queries("x", "{x}", "y", "{y}"), то диспетчер перехватит запрос GET /Q?x=5&y=10 и 
передаст его в handlerQ. Но если клиент пришлет просто GET /Q?x=5 (забыв y), роутер мгновенно отклонит 
запрос с ошибкой 404 Not Found, даже не вызывая вашу функцию. Это идеальный способ валидации входящих 
запросов на уровне роутера.

4. Зачем нужен strconv.ParseFloat? Как вы можете заметить на слайде с примером GET / ? x=4.00, y=4.50
, для выполнения расчетов над параметрами (если мы хотим их сложить или как-то обработать), мы обязаны 
перевести их из текста в числа. Функция strconv.ParseFloat(строка, 64) берет строковый параметр и пытается
 превратить его в тип float64 (вещественное число двойной точности).


 package main

import (
	"fmt"      // Пакет для отправки текстовых ответов обратно в браузер/клиент
	"log"      // Пакет для логирования и вывода системных сообщений в консоль
	"net/http" // Стандартный сетевой пакет Go для работы по протоколу HTTP
	"strconv"  // Пакет для конвертации строк в числа (string -> float64)

	// Подключаем продвинутый маршрутизатор (роутер) Gorilla Mux
	"github.com/gorilla/mux"
)

// handlerQ — функция-обработчик GET-запросов, содержащих query-параметры.
// Ожидает запрос вида: GET http://localhost:3000/Q?x=5.5&y=10.2
func handlerQ(w http.ResponseWriter, r *http.Request) {
	// ШАГ 1: Извлечение сырых строк из URL-адреса.
	// `r.URL.Query()` считывает всё, что идет после знака "?" в ссылке, и преобразует в словарь.
	// Метод `.Get("x")` ищет ключ "x". Если клиент забыл его передать, 
	// метод не сломает программу, а просто вернет пустую строку "".
	xStr := r.URL.Query().Get("x")
	yStr := r.URL.Query().Get("y")

	// ШАГ 2: Сохранение строковых данных в ассоциативный массив (map).
	// Мы создаем временную карту `q`, где ключи и значения — это обычные строки.
	q := map[string]string{
		"x": xStr,
		"y": yStr,
	}

	// Выводим строковые значения обратно клиенту.
	// Спецификатор `%s` используется строго для вставки обычного текста (string).
	fmt.Fprintf(w, "Строковые параметры: x = %s , y = %s\n", q["x"], q["y"])

	// ШАГ 3: Математическая обработка (Конвертация).
	// Любые параметры из URL всегда прилетают на сервер в виде чистого текста.
	// Чтобы с ними можно было считать (складывать, умножать), мы переводим их в числа с плавающей точкой.
	// `strconv.ParseFloat` принимает строку и разрядность (64 бита).
	// Знак нижнего подчеркивания `_` игнорирует возможную ошибку, если вместо числа прилетят буквы.
	x, _ := strconv.ParseFloat(xStr, 64)
	y, _ := strconv.ParseFloat(yStr, 64)

	// Выводим результат числовой обработки.
	// Спецификатор `%.2f` округляет и выводит вещественное число ровно с двумя знаками после запятой.
	fmt.Fprintf(w, "Числовые параметры: x = %.2f, y = %.2f\n", x, y)
}

func main() {
	// Инициализируем маршрутизатор Gorilla Mux
	r := mux.NewRouter()

	// НАСТРОЙКА СТРОГОЙ МАРШРУТИЗАЦИИ (Важная фишка библиотеки gorilla/mux)
	// Метод `.Queries()` работает как строгий сетевой фильтр.
	// Пара `"x", "{x}"` означает: роутер обязан найти в URL ключ с именем "x", а в скобках `{x}` 
	// указано, что его значение может быть абсолютно любым.
	// Если клиент сделает запрос без параметров (просто `/Q`), этот маршрут его проигнорирует 
	// и выдаст ошибку 404 Not Found, так как условия фильтрации не выполнены.
	r.HandleFunc("/Q", handlerQ).
		Methods(http.MethodGet).
		Queries("x", "{x}", "y", "{y}")

	log.Println("Server is running on port 3000")
	log.Println("Test with: GET http://localhost:3000/Q?x=5.5&y=10.2")

	// Запускаем веб-сервер на 3000 порту и отдаем ему наш роутер `r` вместо стандартного `nil`.
	// Если порт занят другим процессом, log.Fatal выведет ошибку и мгновенно закроет приложение.
	if err := http.ListenAndServe(":3000", r); err != nil {
		log.Fatal(err)
	}
}




// 1. Метод r.URL.Query() разбирает строку параметров и возвращает словарь.
	// Метод Get() безопасно извлекает значение по ключу.
	// Если параметр не передан, вернется пустая строка "".
