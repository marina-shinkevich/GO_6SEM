1. Как работают шаблоны маршрутов ({name}) в gorilla/mux? В стандартном net/http маршрутизация статична.
 Пакет gorilla/mux позволяет использовать фигурные скобки {} для указания динамических частей пути 
 (переменных). Когда приходит запрос, например GET /A/Student, роутер понимает, что слово "Student"
  — это значение для шаблона {name}. Внутри функции мы вызываем mux.Vars(r), которая возвращает
   ассоциативный массив (map), и достаем значение по ключу vars["name"]. Дополнительный плюс на экзамене:
    можете упомянуть, что в mux можно использовать регулярные выражения, например /{id:+}, чтобы маршрут 
    срабатывал только если переданы цифры.

2. Что такое формат application/x-www-form-urlencoded? Это стандартный тип кодирования данных, 
который используют обычные веб-страницы при отправке тега <form method="POST">. Данные не вкладываются в URL 
(как при GET-запросах), а передаются в самом теле (body) HTTP-запроса в виде строки, склеенной амперсандами:
 name=Ivan&message=Hello.

 application/x-www-form-urlencoded — стандартный формат передачи данных HTML-формы в теле POST-запроса в 
 виде ключ=значение&ключ=значение. Например: name=Ivan&message=Hello.

3. Как Go обрабатывает эти данные (связка ParseForm и FormValue)?
    Вызов r.ParseForm() заставляет Go вычитать байты из тела запроса и разобрать эту склеенную строку во внутреннюю коллекцию ключей и значений. Если данные пришли битые, функция вернет ошибку.
    Функция r.FormValue("name") безопасно извлекает значение конкретного поля по его имени. Важная 
    особенность метода FormValue в Go заключается в том, что если ParseForm по какой-то причине забыли
     вызвать явно, FormValue попытается вызвать его автоматически (однако в серверном программировании 
     принято делать явный вызов для правильной обработки ошибок, как показано в коде).




     package main

import (
	"encoding/json" // Пакет-конвертер. Умеет превращать данные Go (например, map) в текст JSON и обратно.
	"fmt"           // Пакет для форматирования текста и отправки его обратно в сеть.
	"log"           // Пакет для красивого вывода системных сообщений в консоль (черное окошко) сервера.
	"net/http"      // Базовый сетевой пакет Go для работы по протоколу HTTP (прием и отправка запросов).

	// Подключаем внешний продвинутый маршрутизатор (роутер) Gorilla Mux.
	"github.com/gorilla/mux"
)

// handlerA — функция-обработчик для демонстрации динамических переменных в самом пути URL.
// Она включается, если клиент делает GET-запрос вида: /A/Ivan или /A/123
func handlerA(w http.ResponseWriter, r *http.Request) {
	// ШАГ 1: Потрошим адресную строку.
	// Функция `mux.Vars(r)` сканирует URL-адрес, находит там изменчивые кусочки 
	// и раскладывает их в удобный именованный список (карту строк).
	vars := mux.Vars(r)

	// ШАГ 2: Вытаскиваем значение из списка по имени ячейки.
	// Достаем тот кусочек ссылки, который в функции main() мы назвали словом "name".
	name := vars["name"]

	// ШАГ 3: Отправляем простой текстовый ответ обратно пользователю.
	// Вместо знака `%s` подставится значение переменной `name`.
	fmt.Fprintf(w, "Привет, %s!\n", name)
}

// handlerC — функция-обработчик для POST-запроса, который принимает скрытые данные из тела запроса.
// Она ожидает классическую отправку данных, как из обычной HTML-формы на сайтах.
func handlerC(w http.ResponseWriter, r *http.Request) {
	// ШАГ 1: Парсинг (разбор) прилетевшей формы.
	// По умолчанию данные в теле POST-запроса лежат в виде одной сплошной зашифрованной строки 
	// (например, "name=John&message=Hello"). 
	// Метод `ParseForm()` разбивает эту строку на отдельные аккуратные пары «ключ-значение».
	if err := r.ParseForm(); err != nil {
		// Если данные формы пришли битыми или оборванными, возвращаем ошибку 400 Bad Request
		http.Error(w, "cannot parse form", http.StatusBadRequest)
		return
	}

	// ШАГ 2: Извлекаем конкретные значения по их текстовым ключам.
	// Метод `r.FormValue("name")` ищет в форме поле, которое заполнил пользователь, с именем "name".
	nameParam := r.FormValue("name")
	messageParam := r.FormValue("message")

	// ШАГ 3: Упаковываем данные в ассоциативный массив (map).
	// Мы создаем временную карту `resp`, где ключами и значениями являются обычные строки.
	// Сюда мы берем данные из формы, а также техническую информацию о самом запросе.
	resp := map[string]string{
		"name":        nameParam,    // Имя из формы
		"message":     messageParam, // Сообщение из формы
		"contentType": r.Header.Get("Content-Type"), // Какую кодировку использовал клиент
		"info":        fmt.Sprintf("%s %s", r.Method, r.RequestURI), // Метод (POST) и адрес, куда стучались
	}

	// ШАГ 4: Устанавливаем правильный заголовок ответа (MIME-тип).
	// Мы официально сообщаем браузеру или программе Postman: 
	// «Внимание! Сейчас обратно прилетит не простой текст, а структурированный JSON-документ».
	w.Header().Set("Content-Type", "application/json; charset=utf-8")
	
	// ШАГ 5: Превращаем карту Go в формат JSON и выталкиваем в сеть.
	// `json.NewEncoder(w)` — создает «робота-писателя», подключенного к сетевому ответу `w`.
	// `.Encode(resp)` — берет нашу карту `resp`, на лету переводит её в формат JSON и отправляет клиенту.
	if err := json.NewEncoder(w).Encode(resp); err != nil {
		// Если внутри сервера на этапе кодирования JSON что-то сломалось, отдаем ошибку 500
		http.Error(w, err.Error(), http.StatusInternalServerError)
	}
}

func main() {
	// Создаем главный «пульт управления» маршрутами (роутер) из библиотеки Gorilla Mux.
	r := mux.NewRouter()

	// 1. НАСТРОЙКА МАРШРУТА С ПЕРЕМЕННОЙ ПУТИ (Path Parameter).
	// Фигурные скобки `{name}` создают динамическую ячейку в ссылке. 
	// Роутер перехватит любое слово или число после `/A/` и привяжет этот запрос к функции handlerA.
	// Метод `.Methods(http.MethodGet)` разрешает только обычный просмотр (GET).
	r.HandleFunc("/A/{name}", handlerA).Methods(http.MethodGet)

	// 2. НАСТРОЙКА МАРШРУТА ДЛЯ ПРИЕМА ДАННЫХ ИЗ ФОРМЫ.
	// Этот маршрут сработает, только если клиент делает POST-запрос на точный адрес `/C`.
	// Все данные будут передаваться скрыто внутри тела (Body) запроса.
	r.HandleFunc("/C", handlerC).Methods(http.MethodPost)

	// Выводим в черное окно (консоль) сервера информацию о том, что мы запустились.
	log.Println("Server is running on 3000")
	
	// Включаем сервер на локальном адресе 127.0.0.1 и порту 3000. 
	// Передаем ему наш кастомный роутер `r` вместо стандартного `nil`.
	// Если порт 3000 занят другой программой, `log.Fatal` мгновенно выключит сервер и покажет ошибку.
	if err := http.ListenAndServe("127.0.0.1:3000", r); err != nil {
		log.Fatal(err)
	}
}
