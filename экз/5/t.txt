package main

import (
	"log"
	"net/http"

	"github.com/gorilla/websocket"
)

// 1. Создаем объект Upgrader (обновлятель соединения).
// Он нужен для того, чтобы превратить обычный HTTP-запрос в долгоживущее WebSocket-соединение.
var upgrader = websocket.Upgrader{
	// CheckOrigin проверяет, с какого домена пришел запрос.
	// Возвращая true, мы отключаем защиту CORS и разрешаем подключаться любым клиентам.
	CheckOrigin: func(r *http.Request) bool { return true },
}

// handler — функция обработки веб-сокет соединений
func handler(w http.ResponseWriter, r *http.Request) {
	// 2. Апгрейд соединения (процедура Handshake).
	// Если успешно, мы получаем объект conn (само соединение).
	if conn, err := upgrader.Upgrade(w, r, nil); err == nil {
		defer conn.Close() // Обязательно закрываем соединение при выходе из функции

		// 3. Бесконечный цикл прослушивания входящих сообщений
		for {
			// Читаем тип сообщения (текст/бинарное) и само сообщение (массив байт)
			msgType, msg, err := conn.ReadMessage()

			if err != nil {
				// 4. Корректная обработка отключения клиента
				if websocket.IsCloseError(err, websocket.CloseNormalClosure) {
					log.Println("client closed (1000 normal)") // Код 1000 — нормальное закрытие
				} else {
					log.Println("read error:", err) // Обрыв связи или другая ошибка
				}
				break // Выходим из бесконечного цикла, соединение закроется (через defer)
			}

			// Выводим полученное от клиента сообщение в консоль сервера
			log.Println("Получено:", string(msg))

			// 5. Эхо-ответ: отправляем то же самое сообщение обратно клиенту
			if err = conn.WriteMessage(msgType, msg); err != nil {
				log.Println("write:", err)
				break
			}
		}
	} else {
		log.Println("upgrade error:", err)
	}
}

func main() {
	// (Опционально) Раздача статики для тестирования из браузера, как на слайде лекции
	fs := http.FileServer(http.Dir("./static"))
	http.Handle("/client/", http.StripPrefix("/client/", fs))

	// Регистрируем наш WebSocket обработчик на адрес /ws
	http.HandleFunc("/ws", handler)

	log.Println("WebSocket server is running on :3000")
	log.Fatal(http.ListenAndServe(":3000", nil))
}

// client.go
package main

import (
	"fmt"  // Пакет для красивого вывода сообщений на экран/в консоль
	"log"  // Пакет для логирования ошибок (выводит текст с точным временем сбоя)
	"time" // Пакет для работы со временем (нужен для настройки таймера ожидания)

	// Подключаем ту же библиотеку Gorilla WebSocket, но теперь используем её функции для клиента
	"github.com/gorilla/websocket"
)

func main() {
	// Ссылка на наш WebSocket-сервер. 
	// Вместо привычного "http://" мы используем протокол "ws://", чтобы запустить рукопожатие.
	serverURL := "ws://localhost:3001/ws"
	fmt.Printf("Попытка подключения к %s...\n", serverURL)

	// ШАГ 1: Подключение к серверу (Звонок)
	// Функция Dial отправляет HTTP-запрос серверу с предложением: «Давай перейдем на WebSocket?».
	// Возвращает:
	// - c: объект соединения (*websocket.Conn) — наша "трубка" для отправки и приема сообщений.
	// - _: HTTP-ответ сервера (мы его игнорируем с помощью нижнего подчеркивания).
	// - err: ошибку, если сервер выключен или указан неверный порт.
	c, _, err := websocket.DefaultDialer.Dial(serverURL, nil)
	if err != nil {
		// Если сервер не отвечает, выводим ошибку и немедленно завершаем работу всей программы
		log.Fatal("Ошибка подключения к серверу, проверьте запущен ли он:", err)
	}
	
	// `defer` — это напоминалка для программы. Она говорит: «Когда функция main дойдет до самого конца 
	// и клиент будет закрываться, обязательно выполни команду `c.Close()`, чтобы корректно положить трубку».
	defer c.Close()

	fmt.Println("Подключение установлено! Мы в сети.")

	// Содержимое письма, которое мы хотим отправить
	message := "Привет, Gorilla WebSocket!"
	
	// ШАГ 2: Отправка сообщения
	// `WriteMessage` упаковывает строку в байты и выстреливает ею в открытый канал связи.
	// Параметр `websocket.TextMessage` (равен цифре 1) сообщает серверу, что мы шлем обычный текст.
	// `[]byte(message)` превращает строку в понятный для сетевой карты набор байт.
	err = c.WriteMessage(websocket.TextMessage, []byte(message))
	if err != nil {
		log.Println("Не удалось отправить сообщение, сеть оборвалась:", err)
		return // Прерываем работу функции main
	}
	fmt.Printf("Успешно отправлено: %s\n", message)

	// ШАГ 3: Ожидание ответа от сервера с защитой от зависания
	// `SetReadDeadline` устанавливает временной лимит (дэдлайн). 
	// Мы берем текущее время `time.Now()` и прибавляем к нему 5 секунд.
	// Если сервер "упадет" или зависнет и не пришлет ответ в течение 5 секунд, клиент не будет 
	// висеть в памяти компьютера вечно, а принудительно прервет ожидание с ошибкой таймаута.
	c.SetReadDeadline(time.Now().Add(5 * time.Second))
	
	// `ReadMessage()` блокирует программу и "спит", ожидая фрейм от сервера.
	// Возвращает тип сообщения (игнорируем через `_`), сам ответ в байтах (`response`) и ошибку (`err`).
	_, response, err = c.ReadMessage()
	if err != nil {
		// Сюда мы зайдем, если сервер промолчал больше 5 секунд или внезапно отключился
		log.Println("Ошибка при чтении ответа (возможно, вышел таймаут 5 секунд):", err)
		return
	}
	
	// Переводим массив байт ответа обратно в понятную строку через %s
	fmt.Printf("Успешно получен ответ от сервера: %s\n", response)
}
