Ассоциативный массив (в языке Go он называется map, в других языках — словарь, карта, хэш-таблица) — 
это специальная структура данных (динамическая коллекция), которая хранит информацию в виде пар «ключ 
— значение».

package main // Определяем главный пакет, с которого начинается выполнение любой Go-программы

import (
	"fmt"      // Пакет для форматированного ввода/вывода (в данном случае для отправки текста клиенту)
	"log"      // Пакет для логирования (выводит сообщения в консоль с отметкой даты и времени)
	"net/http" // Стандартный пакет Go для реализации HTTP-клиентов и серверов
)

// trace — вспомогательная функция. На уровне сигнатуры она принимает:
// 1. w http.ResponseWriter — интерфейс для формирования и отправки HTTP-ответа клиенту (браузеру)
// 2. r *http.Request — указатель на структуру, содержащую все данные о входящем HTTP-запросе
func trace(w http.ResponseWriter, r *http.Request) {
	// log.Printf выводит лог на стороне СЕРВЕРА в консоль. 
	// r.Method — это HTTP-метод (GET, POST), r.URL.Path — запрашиваемый путь (например, /A)
	log.Printf("Сервер принял запрос: %s %s\n", r.Method, r.URL.Path)

	// fmt.Fprintf отправляет форматированную строку КЛИЕНТУ (в поток ResponseWriter).
	// Это то, что пользователь увидит на экране своего браузера.
	fmt.Fprintf(w, "Ответ от сервера: Метод: %s, Путь: %s\n", r.Method, r.URL.Path)
}

// handleGetRoot — функция-обработчик (handler) для корневого маршрута (GET "/")
func handleGetRoot(w http.ResponseWriter, r *http.Request) {
	// Делегируем выполнение функции трассировки
	trace(w, r)
}

// handleGetA — обработчик для маршрута "GET /A"
func handleGetA(w http.ResponseWriter, r *http.Request) {
	trace(w, r)
}

// handleGetAB — обработчик для вложенного маршрута "GET /A/B"
func handleGetAB(w http.ResponseWriter, r *http.Request) {
	trace(w, r)
}

// router — кастомный диспетчер (ручной роутер/мультиплексор).
// Он принимает абсолютно все входящие запросы и вручную распределяет их по нужным функциям.
func router(w http.ResponseWriter, r *http.Request) {
	// ПЕРВЫЙ УРОВЕНЬ МАРШРУТИЗАЦИИ: Фильтрация по HTTP-методу.
	// Защищает сервер от обработки некорректных методов (например, POST вместо GET).
	switch r.Method {

	case "GET":
		// ВТОРОЙ УРОВЕНЬ МАРШРУТИЗАЦИИ: Анализ URL-пути внутри разрешенного метода GET.
		switch r.URL.Path {
		case "/":
			handleGetRoot(w, r) // Если путь ровно "/", вызываем соответствующий обработчик
		case "/A":
			handleGetA(w, r)    // Если путь "/A"
		case "/A/B":
			handleGetAB(w, r)   // Если путь "/A/B"
		default:
			// Блок по умолчанию (обработка ошибок маршрутизации):
			// Если совпадений по пути нет, возвращаем стандартный HTTP-статус 404 (Not Found).
			http.NotFound(w, r)
		}

	default:
		// Если метод запроса не GET (например, POST, PUT, DELETE):
		// Отправляем клиенту ошибку 405 (Method Not Allowed) с текстовым пояснением.
		http.Error(w, "Метод не поддерживается сервером", http.StatusMethodNotAllowed)
	}
}

func main() {
	// 1. Регистрация глобального обработчика.
	// Функция http.HandleFunc привязывает путь к обработчику в стандартном маршрутизаторе DefaultServeMux.
	// Так как мы указали корень "/", абсолютно ВСЕ входящие запросы (даже /A, /xyz) 
	// по умолчанию перенаправляются в нашу единую функцию router.
	http.HandleFunc("/", router)

	// 2. Информационное сообщение.
	// Выводим в консоль статус о том, что код успешно скомпилирован, запущен и готов слушать порт.
	log.Println("Сервер успешно запущен на http://localhost:3000")

	// 3. Инициализация и запуск сетевого слушателя.
	// http.ListenAndServe(":3000", nil) — запускает бесконечный цикл веб-сервера на порту 3000.
	// Второй параметр `nil` означает, что используется стандартный DefaultServeMux (куда мы выше привязали наш router).
	//
	// log.Fatal(...) — это обертка: если ListenAndServe вернет ошибку (например, порт 3000 занят 
	// другим приложением), log.Fatal прервет работу программы с вызовом os.Exit(1) и выведет ошибку в консоль.
	log.Fatal(http.ListenAndServe(":3000", nil))
}
